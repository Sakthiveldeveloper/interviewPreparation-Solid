// 14.intersectin of 2 array 
function intersectinArray(arr1, arr2){
  // use filter
// const set2 = new Set(inputValue2);
//   const intersection = inputValue1.filter(item => set2.has(item));
//   return intersection
// use for loop
const intersection = [];

for (let i = 0; i < arr1.length; i++) {
  for (let j = 0; j < arr2.length; j++) {
    if (arr1[i] === arr2[j]) {
      intersection.push(arr1[i]);
      break; // Found a match, move to the next element in arr1
    }
  }

} 
return intersection  
}
