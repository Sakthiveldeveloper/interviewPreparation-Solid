Sliding Window is a **DSA pattern**.

# What is Sliding Window Pattern?

Instead of:

* checking every subarray/substring again and again

We:

* maintain a moving window
* expand/shrink it smartly

This reduces time complexity.

---

Use one moving window.




Time:


# 2️⃣ Dynamic Size Window

Window size changes.

Example:

* Longest Substring Without Repeating Characters
* Minimum Window Substring

Template:

```js id="b8tv8n"
for (let right = 0; right < arr.length; right++) {

    // expand window

    while (window invalid) {

        // shrink window
        left++;
    }

    // update answer
}
```

# Brute Force Method

Check every subarray manually:

```txt id="7c2c5w"
[1,2,3] = 6
[2,3,4] = 9
[3,4,5] = 12
```

Code:

```js id="xt5v5f"
function bruteForce(arr, k) {

    let maxSum = 0;

    for (let i = 0; i <= arr.length - k; i++) {

        let sum = 0;

        for (let j = i; j < i + k; j++) {

            sum += arr[j];
        }

        maxSum = Math.max(maxSum, sum);
    }

    return maxSum;
}
```

# Sliding Window Code

```js id="07qy44"
function slidingWindow(arr, k) {

    let windowSum = 0;

    for (let i = 0; i < k; i++) {
        windowSum += arr[i];
    }

    let maxSum = windowSum;

    for (let i = k; i < arr.length; i++) {

        windowSum =
            windowSum - arr[i - k] + arr[i];

        maxSum = Math.max(maxSum, windowSum);
    }

    return maxSum;
}
```

---

# Complexity

O(n)

Much faster.

# 1️⃣ Fixed Size Sliding Window Template

    Used when:

    * window size always fixed
    * like `k`

    Examples:

    * Maximum Sum Subarray of Size K
    * Average of Subarray of Size K

    ---

    # Template

    ```js id="t17iqi"
    function fixedWindow(arr, k) {

        let left = 0;
        let windowSum = 0;

        for (let right = 0; right < arr.length; right++) {

            // 1. add right element
            windowSum += arr[right];

            // 2. when window size becomes k
            if (right >= k - 1) {

                // 3. calculate answer
                console.log(windowSum);

                // 4. remove left element
                windowSum -= arr[left];

                // 5. move left
                left++;
            }
        }
    }
    ```

    ---

    # Flow

    ```txt id="4owng9"
    Add right
    ↓
    Window size reached?
    ↓
    Calculate answer
    ↓
    Remove left
    ↓
    Move left
    ```

    ---

    # Visualization


    Slide →

    ```txt id="lsvx76"
    1 [2 3 4] 5
    ```

# 2️⃣ Dynamic Sliding Window Template

    Used when:

    * window size changes dynamically

    Examples:

    * Longest Substring Without Repeating Characters
    * Minimum Window Substring
    * Fruits Into Baskets

    ---

    # Template

    ```js id="m6pk19"
    function dynamicWindow(arr) {

        let left = 0;

        for (let right = 0; right < arr.length; right++) {

            // 1. expand window
            console.log(arr[right]);

            // 2. shrink while invalid
            while (windowIsInvalid) {

                left++;
            }

            // 3. calculate answer
        }
    }
    ```

    ---

    # Real Dynamic Template

    ```js id="hhg6f6"
    function longestSubstring(s) {

        let set = new Set();

        let left = 0;
        let maxLen = 0;

        for (let right = 0; right < s.length; right++) {

            // expand
            while (set.has(s[right])) {

                set.delete(s[left]);

                left++;
            }

            set.add(s[right]);

            // calculate answer
            maxLen = Math.max(
                maxLen,
                right - left + 1
            );
        }

        return maxLen;
    }
    ```

    ---

    # Dynamic Window Flow

    ```txt id="90zv7x"
    Expand window
    ↓
    Invalid?
    ↓ yes
    Shrink window
    ↓
    Valid again
    ↓
    Update answer
    ```
