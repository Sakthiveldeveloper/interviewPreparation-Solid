**Q1. What is Node.js? How is it different from browser JavaScript?**

Node.js is a JavaScript runtime built on Chrome's V8 engine that allows running JavaScript on the server side.

```
Browser JavaScript          Node.js
─────────────────          ───────
Runs in browser            Runs on server
Has DOM, BOM               No DOM, BOM
window, document           global, process
Limited file access        Full file system access
XMLHttpRequest/fetch       http, https modules
localStorage               No localStorage
```

**Key features of Node.js:**
- **Non-blocking I/O** — doesn't wait for operations to complete
- **Event-driven** — uses events and callbacks
- **Single-threaded** — one thread handles all requests via Event Loop
- **NPM** — largest package ecosystem

**Real use cases:** REST APIs, real-time apps, microservices, CLI tools

**Q2. Event Loop in Node.js — How does it work?**

```
   ┌──────────────────────────┐
   │        timers            │  ← setTimeout, setInterval
   ├──────────────────────────┤
   │     pending callbacks    │  ← I/O callbacks
   ├──────────────────────────┤
   │       idle, prepare      │  ← internal use
   ├──────────────────────────┤
   │          poll            │  ← fetch new I/O events
   ├──────────────────────────┤
   │          check           │  ← setImmediate
   └──────────────────────────┘

Priority order:
1. process.nextTick()     ← highest priority!
2. Promise callbacks      ← microtasks
3. setTimeout/setInterval ← timers phase
4. setImmediate           ← check phase
```

```javascript
console.log('1 - sync');

setTimeout(() => console.log('2 - setTimeout'), 0);
setImmediate(() => console.log('3 - setImmediate'));
Promise.resolve().then(() => console.log('4 - Promise'));
process.nextTick(() => console.log('5 - nextTick'));

console.log('6 - sync');

// Output: 1, 6, 5, 4, 2, 3
// nextTick > Promise > setTimeout > setImmediate
```

**Q3. CommonJS vs ES Modules?**

```javascript

const { Router } = require('express');

module.exports.myFunction = myFunction;

// ES Modules (modern — use "type": "module" in package.json)

// Key differences:
// CommonJS: require() is synchronous, can be conditional
// ES Modules: import is static, hoisted, async
// CommonJS: __dirname, __filename available
```

**Q4. Streams & Buffers in Node.js?**

```javascript
const fs = require('fs');

// Duplex   — both readable and writable (TCP socket)
```

**Q6. JWT Authentication — How it works?**

```javascript
// JWT = Header.Payload.Signature
// Example: eyJhbGc.eyJ1c2VySW.SflKxwRJS

const jwt = require('jsonwebtoken');
const bcrypt = require('bcrypt');

// LOGIN — generate token
app.post('/login', async (req, res, next) => {
  try {
    const { email, password } = req.body;

    // Find user
    const user = await User.findOne({ email });
    if (!user) throw new AppError('Invalid credentials', 401);

    // Compare password
    const isValid = await bcrypt.compare(password, user.password);
    if (!isValid) throw new AppError('Invalid credentials', 401);

    // Generate tokens
    const accessToken = jwt.sign(
      { userId: user._id, email: user.email, role: user.role },
      process.env.JWT_SECRET,
      { expiresIn: '15m' } // short lived!
    );

    const refreshToken = jwt.sign(
      { userId: user._id },
      process.env.REFRESH_SECRET,
      { expiresIn: '7d' } // long lived
    );

    // Store refresh token in DB
    await user.updateOne({ refreshToken });

    res.json({
      success: true,
      accessToken,
      refreshToken
    });
  } catch(err) {
    next(err);
  }
});

// PROTECT ROUTES — verify token
const authenticate = (req, res, next) => {
  try {
    const token = req.headers.authorization?.split(' ')[1];
    if (!token) throw new AppError('No token provided', 401);

    req.user = jwt.verify(token, process.env.JWT_SECRET);
    next();
  } catch(err) {
    if (err.name === 'TokenExpiredError') {
      return res.status(401).json({ message: 'Token expired' });
    }
    next(new AppError('Invalid token', 401));
  }
};

// REFRESH TOKEN — get new access token
app.post('/refresh', async (req, res, next) => {
  try {
    const { refreshToken } = req.body;
    const decoded = jwt.verify(refreshToken, process.env.REFRESH_SECRET);

    const user = await User.findOne({
      _id: decoded.userId,
      refreshToken
    });
    if (!user) throw new AppError('Invalid refresh token', 401);

    const newAccessToken = jwt.sign(
      { userId: user._id, email: user.email },
      process.env.JWT_SECRET,
      { expiresIn: '15m' }
    );

    res.json({ accessToken: newAccessToken });
  } catch(err) {
    next(err);
  }
});
```

**Q9. Node.js Clustering for Performance?**

```javascript
// Single thread can only use 1 CPU core
// Clustering = run multiple Node.js processes (one per CPU core)

const cluster = require('cluster');
const os = require('os');
const express = require('express');

if (cluster.isPrimary) {
  const numCPUs = os.cpus().length;
  console.log(`Primary process ${process.pid} running`);
  console.log(`Forking ${numCPUs} workers...`);

  // Fork workers (one per CPU core)
  for (let i = 0; i < numCPUs; i++) {
    cluster.fork();
  }

  // Restart crashed workers
  cluster.on('exit', (worker, code, signal) => {
    console.log(`Worker ${worker.process.pid} died`);
    console.log('Starting new worker...');
    cluster.fork(); // restart!
  });

} else {
  // Worker processes run the actual server
  const app = express();

  app.get('/', (req, res) => {
    res.json({ pid: process.pid, message: 'Hello!' });
  });

  app.listen(3000, () => {
    console.log(`Worker ${process.pid} started`);
  });
}

// Result: 8 CPUs = 8 workers = 8x more requests handled!
// This is what you implemented for the 1 lakh Excel sheet processing!
```

**Q10. Caching with Redis in Node.js?**

```javascript
const redis = require('redis');
const client = redis.createClient({
  url: process.env.REDIS_URL
});

client.connect();

// Cache middleware
const cache = (duration) => async (req, res, next) => {
  const key = `cache:${req.url}`;

  try {
    const cached = await client.get(key);
    if (cached) {
      console.log('Cache HIT:', key);
      return res.json(JSON.parse(cached));
    }
    console.log('Cache MISS:', key);

    // Override res.json to cache the response
    const originalJson = res.json.bind(res);
    res.json = async (data) => {
      await client.setEx(key, duration, JSON.stringify(data));
      return originalJson(data);
    };

    next();
  } catch(err) {
    next(); // if Redis fails, continue without cache
  }
};

// Usage — cache for 5 minutes
app.get('/users', cache(300), async (req, res) => {
  const users = await User.find();
  res.json({ success: true, data: users });
});

// Invalidate cache when data changes
app.post('/users', async (req, res) => {
  const user = await User.create(req.body);

  // Clear users cache
  await client.del('cache:/users');

  res.status(201).json({ success: true, data: user });
});

// Cache use cases:
// - Frequently read, rarely changed data
// - Expensive DB queries
// - Third-party API responses
// - Session data
```

**C2. Build request validation middleware:**
```javascript
const Joi = require('joi');

// Generic validation middleware factory
const validate = (schema) => (req, res, next) => {
  const { error, value } = schema.validate(req.body, {
    abortEarly: false, // return ALL errors not just first
    stripUnknown: true // remove unknown fields
  });

  if (error) {
    const errors = error.details.map(detail => ({
      field: detail.path.join('.'),
      message: detail.message.replace(/['"]/g, '')
    }));

    return res.status(400).json({
      success: false,
      message: 'Validation failed',
      errors
    });
  }

  req.body = value; // use sanitized values
  next();
};

// Schema definitions
const schemas = {
  createUser: Joi.object({
    name: Joi.string().min(2).max(50).required(),
    email: Joi.string().email().required(),
    password: Joi.string().min(8).pattern(
      /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)/
    ).required().messages({
      'string.pattern.base': 'Password must contain uppercase, lowercase and number'
    }),
    role: Joi.string().valid('user', 'admin').default('user')
  }),

  updateUser: Joi.object({
    name: Joi.string().min(2).max(50),
    email: Joi.string().email()
  }).min(1) // at least one field required
};

// Usage
const userRouter = require('express').Router();
userRouter.post('/users', validate(schemas.createUser), createUser);
userRouter.patch('/users/:id', validate(schemas.updateUser), updateUser);
```

**C3. Build a centralized error handling system:**
```javascript
// utils/AppError.js
class AppError extends Error {
  constructor(message, statusCode, errors = null) {
    super(message);
    this.statusCode = statusCode;
    this.status = statusCode >= 400 && statusCode < 500 ? 'fail' : 'error';
    this.isOperational = true; // our known errors
    this.errors = errors;
    Error.captureStackTrace(this, this.constructor);
  }
}

// middleware/errorHandler.js
const errorHandler = (err, req, res, next) => {
  err.statusCode = err.statusCode || 500;

  // Log error
  console.error('ERROR:', {
    message: err.message,
    stack: err.stack,
    url: req.url,
    method: req.method
  });

  // Handle specific error types
  if (err.name === 'ValidationError') { // Mongoose validation
    err = new AppError('Validation failed', 400,
      Object.values(err.errors).map(e => ({
        field: e.path,
        message: e.message
      }))
    );
  }

  if (err.code === 11000) { // MongoDB duplicate key
    const field = Object.keys(err.keyValue)[0];
    err = new AppError(`${field} already exists`, 409);
  }

  if (err.name === 'JsonWebTokenError') {
    err = new AppError('Invalid token', 401);
  }

  if (err.name === 'TokenExpiredError') {
    err = new AppError('Token expired', 401);
  }

  // Send response
  res.status(err.statusCode).json({
    success: false,
    message: err.message,
    ...(err.errors && { errors: err.errors }),
    ...(process.env.NODE_ENV === 'development' && { stack: err.stack })
  });
};

module.exports = errorHandler;
```

**C5. Build WebSocket server with Socket.io:**
```javascript
const express = require('express');
const http = require('http');
const { Server } = require('socket.io');
const jwt = require('jsonwebtoken');

const app = express();
const server = http.createServer(app);

const io = new Server(server, {
  cors: { origin: process.env.CLIENT_URL }
});

// Socket.io auth middleware
io.use((socket, next) => {
  try {
    const token = socket.handshake.auth.token;
    socket.user = jwt.verify(token, process.env.JWT_SECRET);
    next();
  } catch {
    next(new Error('Unauthorized'));
  }
});

// Track online users
const onlineUsers = new Map();

io.on('connection', (socket) => {
  console.log(`User ${socket.user.userId} connected`);

  // Add to online users
  onlineUsers.set(socket.user.userId, socket.id);
  io.emit('users:online', Array.from(onlineUsers.keys()));

  // Join room
  socket.on('room:join', (roomId) => {
    socket.join(roomId);
    socket.to(roomId).emit('room:userJoined', {
      userId: socket.user.userId,
      name: socket.user.name
    });
  });

  // Handle message
  socket.on('message:send', async (data) => {
    const { roomId, content } = data;

    // Save to DB
    const message = await Message.create({
      sender: socket.user.userId,
      room: roomId,
      content
    });

    // Emit to everyone in room
    io.to(roomId).emit('message:new', {
      id: message._id,
      sender: socket.user,
      content,
      createdAt: message.createdAt
    });
  });

  // Typing indicator
  socket.on('typing:start', (roomId) => {
    socket.to(roomId).emit('typing:user', {
      userId: socket.user.userId,
      isTyping: true
    });
  });

  // Disconnect
  socket.on('disconnect', () => {
    onlineUsers.delete(socket.user.userId);
    io.emit('users:online', Array.from(onlineUsers.keys()));
  });
});

server.listen(3000);
```

```javascript
    max = 100,
    keyGenerator = (req) => req.ip,
    message = 'Too many requests'
  } = options;

  return async (req, res, next) => {
    const key = `ratelimit:${keyGenerator(req)}`;
    const windowSeconds = Math.floor(windowMs / 1000);

    try {
      const [[, count], [, ttl]] = await client.multi()
        .incr(key)
        .ttl(key)
        .exec();

      // Set expiry on first request
      if (count === 1) {
        await client.expire(key, windowSeconds);
      }

      // Set headers
      res.setHeader('X-RateLimit-Limit', max);
      res.setHeader('X-RateLimit-Remaining', Math.max(0, max - count));
      res.setHeader('X-RateLimit-Reset', Date.now() + (ttl * 1000));

      if (count > max) {
        return res.status(429).json({
          success: false,
          message,
          retryAfter: ttl
        });
      }

      next();
    } catch(err) {
      next(); // if Redis fails, allow request
    }
  };
};

// Usage
app.use('/api', rateLimiter({ windowMs: 15 * 60 * 1000, max: 100 }));
app.use('/api/auth/login', rateLimiter({
  windowMs: 15 * 60 * 1000,
  max: 5,
  keyGenerator: (req) => `${req.ip}:${req.body.email}`
}));
```

**S4. How do you implement refresh token rotation for better security?**

**Answer:**

```javascript
// Refresh token rotation — issue NEW refresh token on every refresh
// Old refresh token becomes invalid immediately!

app.post('/auth/refresh', catchAsync(async (req, res) => {
  const { refreshToken } = req.cookies; // store in httpOnly cookie!

  if (!refreshToken) throw new AppError('No refresh token', 401);

  // Verify token
  const decoded = jwt.verify(refreshToken, process.env.REFRESH_SECRET);

  // Find user with this refresh token
  const user = await User.findOne({
    _id: decoded.userId,
    refreshToken,
    refreshTokenExpiry: { $gt: Date.now() }
  });

  if (!user) {
    // Token reuse detected — possible theft!
    // Invalidate ALL refresh tokens for this user
    await User.updateOne(
      { _id: decoded.userId },
      { $unset: { refreshToken: 1 } }
    );
    throw new AppError('Token reuse detected — please login again', 401);
  }

  // Generate NEW tokens (rotation!)
  const newAccessToken = jwt.sign(
    { userId: user._id, role: user.role },
    process.env.JWT_SECRET,
    { expiresIn: '15m' }
  );

  const newRefreshToken = jwt.sign(
    { userId: user._id },
    process.env.REFRESH_SECRET,
    { expiresIn: '7d' }
  );

  // Save new refresh token (old one invalidated!)
  await user.updateOne({
    refreshToken: newRefreshToken,
    refreshTokenExpiry: Date.now() + 7 * 24 * 60 * 60 * 1000
  });

  // Set in httpOnly cookie (can't be accessed by JS — safer!)
  res.cookie('refreshToken', newRefreshToken, {
    httpOnly: true,
    secure: true,
    sameSite: 'strict',
    maxAge: 7 * 24 * 60 * 60 * 1000
  });

  res.json({ accessToken: newAccessToken });
}));
```

**S8. How do you implement graceful shutdown in Node.js?**

**Answer:**

```javascript
const server = app.listen(PORT);

// Track active connections
let connections = new Set();

server.on('connection', (conn) => {
  connections.add(conn);
  conn.on('close', () => connections.delete(conn));
});

// Graceful shutdown handler
const gracefulShutdown = async (signal) => {
  logger.info(`${signal} received — shutting down gracefully`);

  // Stop accepting new connections
  server.close(async () => {
    logger.info('HTTP server closed');

    try {
      // Close existing connections
      connections.forEach(conn => conn.destroy());

      // Close database connections
      await mongoose.connection.close();
      logger.info('MongoDB connection closed');

      // Close Redis connection
      await redisClient.quit();
      logger.info('Redis connection closed');

      // Complete pending queue jobs
      await emailQueue.close();
      logger.info('Bull queues closed');

      logger.info('Graceful shutdown complete');
      process.exit(0);
    } catch(err) {
      logger.error('Error during shutdown:', err);
      process.exit(1);
    }
  });

  // Force exit after 30 seconds
  setTimeout(() => {
    logger.error('Could not close connections in time — forcing shutdown');
    process.exit(1);
  }, 30000);
};

// Handle shutdown signals
process.on('SIGTERM', () => gracefulShutdown('SIGTERM')); // Docker stop
process.on('SIGINT', () => gracefulShutdown('SIGINT'));   // Ctrl+C

// Handle uncaught errors
process.on('uncaughtException', (err) => {
  logger.error('Uncaught exception:', err);
  gracefulShutdown('uncaughtException');
});

process.on('unhandledRejection', (reason, promise) => {
  logger.error('Unhandled rejection:', { reason, promise });
  gracefulShutdown('unhandledRejection');
});
```

**S10. Senior/Lead question: How do you design a Node.js backend for a system that processes payments?**

**Answer:**

```javascript
// Payment system design principles:
// 1. Idempotency — same request can be safely retried
// 2. Atomicity — either full success or full rollback
// 3. Audit trail — log every state change
// 4. Security — PCI DSS compliance

// Idempotency key — prevent duplicate charges
app.post('/payments', authenticate, catchAsync(async (req, res) => {
  const idempotencyKey = req.headers['idempotency-key'];
  if (!idempotencyKey) throw new AppError('Idempotency key required', 400);

  // Check if request was already processed
  const existing = await PaymentLog.findOne({ idempotencyKey });
  if (existing) {
    return res.json({ success: true, data: existing.result });
  }

  // Process payment in transaction
  const session = await mongoose.startSession();
  session.startTransaction();

  try {
    const { amount, currency, userId, orderId } = req.body;

    // 1. Create pending payment record
    const payment = await Payment.create([{
      userId, orderId, amount, currency,
      status: 'pending',
      idempotencyKey
    }], { session });

    // 2. Call payment gateway (Stripe)
    const stripePayment = await stripe.paymentIntents.create({
      amount: amount * 100, // cents
      currency,
      metadata: { orderId, paymentId: payment[0]._id.toString() }
    });

    // 3. Update payment with gateway reference
    await Payment.updateOne(
      { _id: payment[0]._id },
      { gatewayRef: stripePayment.id, status: 'processing' },
      { session }
    );

    // 4. Log for audit
    await PaymentLog.create([{
      paymentId: payment[0]._id,
      idempotencyKey,
      event: 'payment_initiated',
      result: stripePayment
    }], { session });

    await session.commitTransaction();

    res.status(201).json({
      success: true,
      data: {
        paymentId: payment[0]._id,
        clientSecret: stripePayment.client_secret
      }
    });
  } catch(err) {
    await session.abortTransaction();
    throw err;
  } finally {
    session.endSession();
  }
}));

// Webhook — handle payment completion
app.post('/webhooks/stripe',
  express.raw({ type: 'application/json' }), // raw body for signature verification
  async (req, res) => {
    const sig = req.headers['stripe-signature'];

    let event;
    try {
      event = stripe.webhooks.constructEvent(
        req.body, sig, process.env.STRIPE_WEBHOOK_SECRET
      );
    } catch {
      return res.status(400).send('Invalid signature');
    }

    if (event.type === 'payment_intent.succeeded') {
      const { orderId, paymentId } = event.data.object.metadata;

      await Payment.updateOne({ _id: paymentId }, { status: 'completed' });
      await Order.updateOne({ _id: orderId }, { status: 'paid' });

      await emailQueue.add({ type: 'payment-success', orderId });
```
